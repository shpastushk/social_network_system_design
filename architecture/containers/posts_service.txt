@startuml
!include <C4/C4_Container>

title Posts System Containers Diagram for Travel Social Media

Person(User, "User", "A user of the system")

Container(APIGateway, "API Gateway", "Proxy for users requests")

System_Boundary(system, "Posts system") {
   Container(PostsService, "Posts service", "Go", "Process requests to posts")
   ContainerDb(PostsDB, "Posts Database", "Postgis", "Stores information about posts/comments/likes")
   Container(FeedService, "Feed Service", "Go", "Process data about users and their posts")
   ContainerDb(TarantoolDB, "Feed Database", "Tarantool", "Stores information about users and their posts")
   Container(Kafka, "Kafka")
}

Container(MediaSystem, "Media system", "Software system", "Process media queries")
Container(KafkaUser, "Kafka")
Container(UsersSystem, "Users system", "Software system", "Process users queries")


Rel(User, APIGateway, "REST")
Rel(APIGateway, PostsService, "Makes posts queries", "REST")
Rel(PostsService, PostsDB, "Create update get posts/likes/comments")
Rel(FeedService, TarantoolDB, "Save data about users posts")
Rel(PostsService, Kafka, "Info about new/delete posts")
Rel(FeedService, Kafka, "Get data about posts")
Rel(APIGateway, FeedService, "Queries")
Rel(MediaSystem, Kafka, "Get data about deleted posts")
Rel(UsersSystem, KafkaUser, "Data about changed relationthips between users")
Rel(FeedService, KafkaUser, "Info about changed users relationships for feed service (changed feed posts for user)")

@enduml
