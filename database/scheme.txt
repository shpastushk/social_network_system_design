// Use DBML to define your database structure
// Docs: https://dbml.dbdiagram.io/docs

Table subscribers {
  subscribing_user_id integer
  subscribed_user_id integer
  created_at timestamp
}

Table users {
  id integer [primary key]
  username varchar
}

Table comments {
  id integer [primary key]
	user_id integer [not null]
	text string [not null]
	published_at timestamp [not null]
	post_id integer [not null]
}

Table posts {
  id integer [primary key]
	text string [not null]
	photo_urls string[]
  user_id integer [not null]
	point geometry.Point
	published_at timestamp [not null]
}

Table likes {
  post_id integer [not null]
  user_id integer [not null]
  is_like bool [not null]
  created_at timestamp [not null]
}

Ref user_posts: posts.user_id > users.id // many-to-one

Ref user_comments: comments.user_id > users.id // many-to-one

Ref post_comments: comments.post_id > posts.id // many-to-one

Ref user_likes: likes.user_id > users.id // many-to-one

Ref post_likes: likes.post_id > posts.id // many-to-one

Ref: users.id < subscribers.subscribing_user_id

Ref: users.id < subscribers.subscribed_user_id
